cmake_minimum_required(VERSION 3.31)
project(iiSDKDeviceCompositionConsumer LANGUAGES CXX)
option(DEVICE_FIRST "Import the independent device package first" ON)
if(DEVICE_FIRST)
    find_package(iiDeviceMultimedia 0.2 CONFIG REQUIRED)
    if(TARGET iiSDKCore::iiSDKCore OR TARGET iiSDKCore::Values)
        message(FATAL_ERROR "Device package imported Core")
    endif()
    find_package(iiSDKCore 0.12 CONFIG REQUIRED)
else()
    find_package(iiSDKCore 0.12 CONFIG REQUIRED)
    if(TARGET iiDeviceMultimedia::iiDeviceMultimedia)
        message(FATAL_ERROR "Core package imported devices")
    endif()
    find_package(iiDeviceMultimedia 0.2 CONFIG REQUIRED)
endif()
add_executable(iiSDKDeviceComposition main.cpp core_first.cpp device_first.cpp)
target_link_libraries(iiSDKDeviceComposition PRIVATE
    iiSDKCore::iiSDKCore iiDeviceMultimedia::iiDeviceMultimedia)
enable_testing()
add_test(NAME iiDeviceMultimedia.Composition COMMAND iiSDKDeviceComposition)
set_tests_properties(iiDeviceMultimedia.Composition PROPERTIES TIMEOUT 30)
if(APPLE)
    set_tests_properties(iiDeviceMultimedia.Composition PROPERTIES
        ENVIRONMENT "OBJC_DEBUG_MISSING_POOLS=YES" FAIL_REGULAR_EXPRESSION "MISSING POOLS")
endif()
