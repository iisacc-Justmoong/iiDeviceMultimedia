cmake_minimum_required(VERSION 3.31)
project(iiDeviceMultimedia VERSION 0.2.1 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
include(cmake/ArchitecturePolicy.cmake)
option(BUILD_SHARED_LIBS "Build shared iiDeviceMultimedia libraries" ON)
option(IIDEVICEMULTIMEDIA_BUILD_EXAMPLES "Build the native device inventory example" ON)
option(IIDEVICEMULTIMEDIA_WASM_THREADS "Enable Emscripten pthread workers" ON)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(BUILD_SHARED_LIBS OFF)
endif()
find_package(Threads REQUIRED)

set(_device_headers IO/Hardware.h IO/Protocols.h IO/Result.h)
foreach(_header IN LISTS _device_headers)
    configure_file("src/${_header}" "${CMAKE_CURRENT_BINARY_DIR}/include/iiDeviceMultimedia/${_header}" COPYONLY)
endforeach()
add_library(iiDeviceMultimedia src/IO/Hardware.cpp src/IO/Protocols.cpp)
add_library(iiDeviceMultimedia::iiDeviceMultimedia ALIAS iiDeviceMultimedia)
target_compile_features(iiDeviceMultimedia PUBLIC cxx_std_23)
target_link_libraries(iiDeviceMultimedia PUBLIC Threads::Threads)
if(EMSCRIPTEN)
    target_compile_options(iiDeviceMultimedia PUBLIC -fexceptions)
    target_link_options(iiDeviceMultimedia PUBLIC -fexceptions)
    if(IIDEVICEMULTIMEDIA_WASM_THREADS)
        target_compile_options(iiDeviceMultimedia PUBLIC -pthread)
        target_link_options(iiDeviceMultimedia PUBLIC -pthread)
    endif()
endif()
target_include_directories(iiDeviceMultimedia PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(iiDeviceMultimedia PROPERTIES CXX_EXTENSIONS OFF
    VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
    target_compile_options(iiDeviceMultimedia PRIVATE /utf-8)
endif()

set(IIDEVICEMULTIMEDIA_PACKAGE_LINUX_IO OFF)
if(APPLE)
    enable_language(OBJCXX)
    set_target_properties(iiDeviceMultimedia PROPERTIES OBJCXX_STANDARD 23
        OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
    target_sources(iiDeviceMultimedia PRIVATE src/IO/HardwareApple.mm)
    set_source_files_properties(src/IO/HardwareApple.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc;-fblocks")
    foreach(_framework Foundation CoreBluetooth AudioToolbox CoreMIDI)
        find_library(_device_${_framework} ${_framework} REQUIRED)
        target_link_libraries(iiDeviceMultimedia PRIVATE ${_device_${_framework}})
    endforeach()
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        find_library(_device_AVFAudio AVFAudio REQUIRED)
        target_link_libraries(iiDeviceMultimedia PRIVATE ${_device_AVFAudio})
    else()
        foreach(_framework IOKit IOUSBHost CoreAudio)
            find_library(_device_${_framework} ${_framework} REQUIRED)
            target_link_libraries(iiDeviceMultimedia PRIVATE ${_device_${_framework}})
        endforeach()
    endif()
elseif(WIN32)
    target_sources(iiDeviceMultimedia PRIVATE src/IO/HardwareWindows.cpp)
    target_compile_definitions(iiDeviceMultimedia PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(iiDeviceMultimedia PRIVATE setupapi cfgmgr32 hid winusb ole32 uuid runtimeobject)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(IIDEVICEMULTIMEDIA_LINUX_IO REQUIRED IMPORTED_TARGET libudev dbus-1 alsa>=1.2.10)
    target_sources(iiDeviceMultimedia PRIVATE src/IO/HardwareLinux.cpp)
    target_link_libraries(iiDeviceMultimedia PRIVATE PkgConfig::IIDEVICEMULTIMEDIA_LINUX_IO)
    set(IIDEVICEMULTIMEDIA_PACKAGE_LINUX_IO ON)
else()
    target_sources(iiDeviceMultimedia PRIVATE src/IO/HardwarePortable.cpp)
endif()

set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiDeviceMultimedia")
install(TARGETS iiDeviceMultimedia EXPORT iiDeviceMultimediaTargets
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
foreach(_header IN LISTS _device_headers)
    get_filename_component(_directory "${_header}" DIRECTORY)
    install(FILES "src/${_header}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/iiDeviceMultimedia/${_directory}")
endforeach()
install(EXPORT iiDeviceMultimediaTargets NAMESPACE iiDeviceMultimedia:: DESTINATION ${_package_dir})
configure_package_config_file(cmake/iiDeviceMultimediaConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiDeviceMultimediaConfig.cmake" INSTALL_DESTINATION ${_package_dir})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiDeviceMultimediaConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiDeviceMultimediaConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/iiDeviceMultimediaConfigVersion.cmake" DESTINATION ${_package_dir})
install(FILES README.md DESTINATION "${CMAKE_INSTALL_DATADIR}/iiDeviceMultimedia")
install(DIRECTORY docs/ DESTINATION "${CMAKE_INSTALL_DATADIR}/iiDeviceMultimedia/docs")

file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/device-boundary-$<CONFIG>.txt"
    CONTENT "$<TARGET_PROPERTY:iiDeviceMultimedia,LINK_LIBRARIES>\n$<TARGET_PROPERTY:iiDeviceMultimedia,INTERFACE_LINK_LIBRARIES>\n$<TARGET_PROPERTY:iiDeviceMultimedia,INCLUDE_DIRECTORIES>\n")
if(BUILD_TESTING)
    add_test(NAME iiDeviceMultimedia.Independence
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/independence.py"
            --source "${CMAKE_CURRENT_SOURCE_DIR}" --build "${CMAKE_CURRENT_BINARY_DIR}")
    foreach(_test Hardware HardwareState Protocols Lifecycle)
        if(_test STREQUAL "HardwareState")
            set(_source hardware_state)
        else()
            string(TOLOWER "${_test}" _source)
        endif()
        add_executable(iiDeviceMultimedia${_test}Tests "tests/${_source}.cpp")
        target_link_libraries(iiDeviceMultimedia${_test}Tests PRIVATE iiDeviceMultimedia::iiDeviceMultimedia)
        add_test(NAME iiDeviceMultimedia.${_test} COMMAND iiDeviceMultimedia${_test}Tests)
        set_tests_properties(iiDeviceMultimedia.${_test} PROPERTIES TIMEOUT 30)
    endforeach()
    if(WIN32)
        target_link_libraries(iiDeviceMultimediaHardwareTests PRIVATE ole32)
    endif()
    if(APPLE)
        set_tests_properties(iiDeviceMultimedia.Hardware PROPERTIES
            ENVIRONMENT "OBJC_DEBUG_MISSING_POOLS=YES" FAIL_REGULAR_EXPRESSION "MISSING POOLS")
    endif()
    set(_header_checks)
    foreach(_header IN LISTS _device_headers)
        string(MAKE_C_IDENTIFIER "${_header}" _name)
        set(_file "${CMAKE_CURRENT_BINARY_DIR}/header-checks/${_name}.cpp")
        file(WRITE "${_file}" "#include <iiDeviceMultimedia/${_header}>\n")
        list(APPEND _header_checks "${_file}")
    endforeach()
    add_library(iiDeviceMultimediaPublicHeaders OBJECT ${_header_checks})
    target_link_libraries(iiDeviceMultimediaPublicHeaders PRIVATE iiDeviceMultimedia::iiDeviceMultimedia)
endif()
if(IIDEVICEMULTIMEDIA_BUILD_EXAMPLES)
    add_executable(iiDeviceMultimediaHardwareExample examples/hardware.cpp)
    target_link_libraries(iiDeviceMultimediaHardwareExample PRIVATE iiDeviceMultimedia::iiDeviceMultimedia)
endif()
if(EMSCRIPTEN AND IIDEVICEMULTIMEDIA_WASM_THREADS)
    foreach(_program iiDeviceMultimediaHardwareTests iiDeviceMultimediaHardwareStateTests)
        if(TARGET ${_program})
            target_link_options(${_program} PRIVATE "-sPROXY_TO_PTHREAD=1"
                "-sPTHREAD_POOL_SIZE=4" "-sEXIT_RUNTIME=1" "-sALLOW_BLOCKING_ON_MAIN_THREAD=0")
        endif()
    endforeach()
endif()
# Keep the extracted native bridges covered by the same source/preprocessor policy.
iidevicemultimedia_enforce_architecture()
