cmake_minimum_required(VERSION 3.31)
project(iiDeviceMultimediaInstalledConsumer LANGUAGES CXX)
find_package(iiDeviceMultimedia 0.2 CONFIG REQUIRED)
if(TARGET iiSDKCore::iiSDKCore OR TARGET iiSDKCore::Values)
    message(FATAL_ERROR "Standalone device package imported another SDK")
endif()
enable_testing()
add_executable(iiDeviceMultimediaInstalledConsumer main.cpp)
target_link_libraries(iiDeviceMultimediaInstalledConsumer PRIVATE iiDeviceMultimedia::iiDeviceMultimedia)
add_test(NAME iiDeviceMultimedia.InstalledConsumer COMMAND iiDeviceMultimediaInstalledConsumer)
foreach(_test Hardware Protocols)
    string(TOLOWER "${_test}" _source)
    add_executable(iiDeviceMultimediaInstalled${_test} "../${_source}.cpp")
    target_link_libraries(iiDeviceMultimediaInstalled${_test} PRIVATE iiDeviceMultimedia::iiDeviceMultimedia)
    add_test(NAME iiDeviceMultimedia.Installed${_test} COMMAND iiDeviceMultimediaInstalled${_test})
    set_tests_properties(iiDeviceMultimedia.Installed${_test} PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
    target_link_libraries(iiDeviceMultimediaInstalledHardware PRIVATE ole32)
endif()
if(APPLE)
    set_tests_properties(iiDeviceMultimedia.InstalledHardware PROPERTIES
        ENVIRONMENT "OBJC_DEBUG_MISSING_POOLS=YES" FAIL_REGULAR_EXPRESSION "MISSING POOLS")
endif()
